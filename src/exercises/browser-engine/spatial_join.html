The Spatial Query

-- 1. Create points from lon/lat
-- 2. Join with neighborhood polygon
SELECT id, category, score 
FROM places, neighborhood
WHERE ST_Intersects(
    ST_Point(places.lon, places.lat), 
    neighborhood.geom
);
